cmake_minimum_required(VERSION 3.13)
project(sfpad CXX)
if(NOT WIN32)
    message(FATAL_ERROR "sfpad requires Windows")
endif()
set(HASHLINK_DIR "" CACHE PATH "HashLink SDK containing include/hl.h")
add_library(sfpad SHARED sfpad.cpp)
target_include_directories(sfpad PRIVATE "${HASHLINK_DIR}/include" "${HASHLINK_DIR}/src")
set_target_properties(sfpad PROPERTIES PREFIX "" SUFFIX ".hdll")
target_compile_features(sfpad PRIVATE cxx_std_17)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "sfpad requires a 64-bit build")
endif()
include(CTest)
if(BUILD_TESTING)
    add_library(probe_sdl3 SHARED tests/mock_sdl3.cpp)
    set_target_properties(probe_sdl3 PROPERTIES OUTPUT_NAME SDL3)
    add_library(probe_engine SHARED tests/mock_engine.cpp)
    target_link_libraries(probe_engine PRIVATE probe_sdl3)
    set_target_properties(probe_engine PROPERTIES OUTPUT_NAME sdl SUFFIX ".hdll")
    add_library(probe_xinput SHARED tests/mock_xinput.cpp)
    set_target_properties(probe_xinput PROPERTIES OUTPUT_NAME XINPUT9_1_0)
    add_library(probe_directx SHARED tests/mock_directx.cpp)
    target_link_libraries(probe_directx PRIVATE probe_xinput)
    set_target_properties(probe_directx PROPERTIES OUTPUT_NAME directx SUFFIX ".hdll")
    add_executable(sfpad_test tests/sfpad_test.cpp)
    target_include_directories(sfpad_test PRIVATE "${HASHLINK_DIR}/include" "${HASHLINK_DIR}/src")
    target_compile_features(sfpad_test PRIVATE cxx_std_17)
    add_dependencies(sfpad_test probe_engine probe_directx)
    add_test(NAME sfpad_passthrough_and_lifecycle COMMAND sfpad_test)
    set_tests_properties(sfpad_passthrough_and_lifecycle PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:sfpad_test>")
    add_test(NAME sfpad_xinput_passthrough_and_lifecycle COMMAND sfpad_test --xinput)
    set_tests_properties(sfpad_xinput_passthrough_and_lifecycle PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:sfpad_test>")
endif()
